cmake_minimum_required(VERSION 3.20)
project(solob_grading CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(grading src/grading.cpp)
target_include_directories(grading PUBLIC include)
add_executable(solob-grade src/cli.cpp)
target_link_libraries(solob-grade grading)
enable_testing()
add_executable(grading_tests tests/test_grading.cpp third_party/catch_amalgamated.cpp)
target_include_directories(grading_tests PRIVATE third_party)
target_link_libraries(grading_tests grading)
add_test(NAME grading_tests COMMAND grading_tests)
